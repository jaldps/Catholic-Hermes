CORONA DE CONFIANZA EN GRANDES TRIBULACIONES
grandes tribulaciones o presagios de peligros por venir.
Se comienza con El Credo (una vez, sólo al comenzar).
Primer Misterio: Él me librará de la red del cazador
Padrenuestro.
Cuantas pequeñas (9 veces) “Eres mi fortaleza y mi refugio, eres mi Dios en quien confío”
Gloria.
Segundo Misterio: Él me librará de la peste mortal
Padrenuestro.
Cuantas pequeñas (9 veces) “Eres mi fortaleza y mi refugio, eres mi Dios en quien confío”
Gloria.
Tercer misterio: Él me cobijará bajo Sus alas y yo me refugiaré bajo sus plumas.
Padrenuestro.
Cuantas pequeñas (9 veces) “Eres mi fortaleza y mi refugio, eres mi Dios en quien confío”
Gloria.
Cuarto misterio: No temeré el terror de la noche, ni la flecha que vuela en el día, ni la peste que avanza en las tinieblas, ni el azote que asola el mediodía.
Padrenuestro.
Cuantas pequeñas (9 veces) “Eres mi fortaleza y mi refugio, eres mi Dios en quien confío”
Gloria.
Quinto misterio: Aunque a mi lado caigan mil y diez mil a mi diestra, a mí no me alcanzarán.
Padrenuestro.
Cuantas pequeñas (9 veces) “Eres mi fortaleza y mi refugio, eres mi Dios en quien confío”
Gloria.
Sexto misterio:  No me alcanzará la desgracia, ni la plaga llegará a mi tienda.
Padrenuestro.
Cuantas pequeñas (9 veces) “Eres mi fortaleza y mi refugio, eres mi Dios en quien confío”
Gloria.
Séptimo misterio:  Él ordenó a sus ángeles, que me guardaran en todos mis caminos.
Padrenuestro.
Cuantas pequeñas (9 veces) “Eres mi fortaleza y mi refugio, eres mi Dios en quien confío”
Gloria.
Octavo misterio: Porque yo me he unido a mi Dios, Él me librará.
Padrenuestro.
Cuantas pequeñas (9 veces) “Eres mi fortaleza y mi refugio, eres mi Dios en quien confío”
Gloria.
Noveno misterio: Me protegerá, porque conozco Su nombre.
Padrenuestro.
Cuantas pequeñas (9 veces) “Eres mi fortaleza y mi refugio, eres mi Dios en quien confío”
Gloria.
Décimo misterio: Él me responderá y estará conmigo en la desgracia, me librará y me llenará de honores, me dará una larga vida y me hará gozar de Su salvación.
Padrenuestro.
Cuantas pequeñas (9 veces) “Eres mi fortaleza y mi refugio, eres mi Dios en quien confío”
Gloria.
Cuando se termina la corona se dice:
a) Un Padrenuestro.
b) “Eres mi fortaleza y mi refugio, eres mi Dios en quien confío” (una vez).
c) Un Gloria.
d) Amén, Amén, Amén.
De esta forma se completan 91.

(function() {
'use strict';
var linkSelector = 'a[aria-controls^="wpml-ls-submenu-"]';
var dismissed = 'wpml-ls-submenu-dismissed';
var dismissible = 'wpml-ls-dismissible-submenu';
function isEscape(e) {
return e.key === 'Escape' || e.key === 'Esc' || e.keyCode === 27;
}
function isShown(submenu) {
var style = window.getComputedStyle(submenu);
if (style.display === 'none' || style.visibility === 'hidden' || parseFloat(style.opacity) === 0) {
return false;
}
var box = submenu.getBoundingClientRect();
return box.width >= 2 && box.height >= 2 && box.right > 0 && box.left < document.documentElement.clientWidth;
}
function uniqueId(base, submenu) {
var id = base, n = 2, element = document.getElementById(id);
while (element && element !== submenu) {
id = base + '-' + n;
n++;
element = document.getElementById(id);
}
return id;
}
function findSubmenu(li) {
for (var child = li.firstElementChild; child; child = child.nextElementSibling) {
if (child.tagName === 'UL') {
return child;
}
}
return li.querySelector('ul');
}
function setUp(li, link) {
var submenu = findSubmenu(li);
if (!submenu) {
link.removeAttribute('aria-expanded');
link.removeAttribute('aria-controls');
link.removeAttribute('role');
return;
}
if (!submenu.id) {
submenu.id = uniqueId(link.getAttribute('aria-controls'), submenu);
}
link.setAttribute('aria-controls', submenu.id);
// The style hides this list (also when it is not a direct child of the item).
submenu.classList.add(dismissible);
var shownAtPointerDown = null;
var sync = function() {
var expanded = isShown(submenu) ? 'true' : 'false';
if (link.getAttribute('aria-expanded') !== expanded) {
link.setAttribute('aria-expanded', expanded);
}
// A theme script may (re)name the submenu after this set-up.
if (submenu.id && link.getAttribute('aria-controls') !== submenu.id) {
link.setAttribute('aria-controls', submenu.id);
}
};
// One pending measure per item: in the next frame, and 350 ms after the last event.
var frame = 0, timer = 0;
var syncSoon = function() {
if (!frame) {
frame = window.requestAnimationFrame(function() {
frame = 0;
sync();
});
}
window.clearTimeout(timer);
timer = window.setTimeout(function() {
timer = 0;
sync();
}, 350);
};
var toggle = function(wasShown) {
if (wasShown) {
li.classList.add(dismissed);
} else {
li.classList.remove(dismissed);
if (document.activeElement !== link) {
link.focus();
}
}
syncSoon();
};
link.addEventListener('pointerdown', function() {
shownAtPointerDown = isShown(submenu);
});
link.addEventListener('click', function(e) {
var handledByTheme = e.defaultPrevented;
var wasShown = shownAtPointerDown === null ? isShown(submenu) : shownAtPointerDown;
shownAtPointerDown = null;
e.preventDefault();
if (handledByTheme) {
// The theme opens and closes this submenu: it decides, not an earlier dismiss.
li.classList.remove(dismissed);
syncSoon();
return;
}
toggle(wasShown);
});
// Another control in the item (a theme's own arrow button) shows or hides the
// submenu: an earlier dismiss must not keep it hidden.
li.addEventListener('click', function(e) {

syncSoon();
}
});

li.classList.remove(dismissed);
}
syncSoon();
});
li.addEventListener('mouseenter', syncSoon);
li.addEventListener('mouseleave', function() {
if (!li.contains(document.activeElement)) {
li.classList.remove(dismissed);
}
syncSoon();
});
submenu.addEventListener('transitionend', sync);
if (window.MutationObserver) {
new MutationObserver(syncSoon).observe(li, { attributes: true, attributeFilter: ['class', 'style', 'hidden', 'aria-expanded', 'id'], subtree: true });
}
sync();
syncSoon();
}

for (var i = 0; i < links.length; i++) {
